	private bool GetMousePositionOnGround(
		out Vector3 position)
	{
		Vector2 mousePosition =
			GetViewport().GetMousePosition();

		Vector3 origin =
			camera.ProjectRayOrigin(mousePosition);

		Vector3 direction =
			camera.ProjectRayNormal(mousePosition);

		if (Mathf.Abs(direction.Y) < 0.001f)
		{
			position = Vector3.Zero;
			return false;
		}

		float distance = -origin.Y / direction.Y;

		if (distance < 0)
		{
			position = Vector3.Zero;
			return false;
		}

		position = origin + direction * distance;
		position.Y = 0;

		return true;
	}

	private void UpdatePreviewMarker()

			segedPont[index].Visible = true;
		}
	}

	private void UpdateVisuals()
	{
		// Az aktuális Bézier-kontrollpontok
		Vector3 p0;
		Vector3 p1;
		Vector3 p2;
		Vector3 p3;

		if (kattAllapot == 0)
		{
			sinVas.Mesh = null;
			sinTalpfa.Mesh = null;
			sinVonalak.Mesh = null;
			return;
		}

		p0 = gorbe[0];

		if (kattAllapot == 1)
		{
			p1 = elonezetPozicio;
			p2 = elonezetPozicio;
			p3 = elonezetPozicio;
		}
		else if (kattAllapot == 2)
		{
			p1 = gorbe[1];
			p2 = elonezetPozicio;
			p3 = elonezetPozicio;
		}
		else if (kattAllapot == 3)
		{
			p1 = gorbe[1];
			p2 = elonezetPozicio;
			p3 = gorbe[2];
		}
		else
		{
			p1 = gorbe[1];
			p2 = gorbe[3];
			p3 = gorbe[2];
		}

		// A görbe mintavételezése
		var curvegorbe = new List<Vector3>();

		const int segments = 100;

		for (int i = 0; i <= segments; i++)
		{
			float t = (float)i / segments;
			float u = 1.0f - t;

			Vector3 point =
				u * u * u * p0 +
				3.0f * u * u * t * p1 +
				3.0f * u * t * t * p2 +
				t * t * t * p3;

			curvegorbe.Add(point);
		}

		// Két sínszál
		var rails = new SurfaceTool();
		rails.Begin(Mesh.PrimitiveType.Triangles);

		// Keresztaljak
		var sleepers = new SurfaceTool();
		sleepers.Begin(Mesh.PrimitiveType.Triangles);

		float distanceSinceSleeper = TalpfaTav;

		for (int i = 0; i < curvegorbe.Count - 1; i++)
		{
			Vector3 a = curvegorbe[i];
			Vector3 b = curvegorbe[i + 1];

			Vector3 difference = b - a;
			float length = difference.Length();

			if (length < 0.0001f)
				continue;

			Vector3 tangent = difference.Normalized();

			// A sínre merőleges vízszintes irány
			Vector3 side =
				Vector3.Up.Cross(tangent).Normalized();

			if (side.LengthSquared() < 0.001f)
				continue;

			// A két sínszál középpontja
			Vector3 center =
				(a + b) * 0.5f +
				Vector3.Up * SinMagassagaTalajtol;

			Vector3 leftRail =
				center - side * (Nyomtav * 0.5f);

			Vector3 rightRail =
				center + side * (Nyomtav * 0.5f);

			// A BoxMesh helyi Z tengelye a sín irányába mutat.
			// Kicsit átfednek a szakaszok, hogy ne legyen rés.
			float railLength = length * 1.04f;

			var leftBasis = new Basis(
				side,
				Vector3.Up,
				tangent * railLength
			);

			var rightBasis = new Basis(
				side,
				Vector3.Up,
				tangent * railLength
			);

			rails.AppendFrom(
				railBox,
				0,
				new Transform3D(leftBasis, leftRail)
			);

			rails.AppendFrom(
				railBox,
				0,
				new Transform3D(rightBasis, rightRail)
			);

			// Keresztaljak egyenletes távolságra
			distanceSinceSleeper += length;

			if (distanceSinceSleeper >= TalpfaTav)
			{
				distanceSinceSleeper = 0;

				Vector3 sleeperCenter =
					center -
					Vector3.Up *
					(SinMagassagaTalajtol - TalpfaMagassag * 0.5f);

				var sleeperBasis = new Basis(
					side,
					Vector3.Up,
					tangent
				);

				sleepers.AppendFrom(
					sleeperBox,
					0,
					new Transform3D(
						sleeperBasis,
						sleeperCenter
					)
				);
			}
		}

		sinVas.Mesh = rails.Commit();
		sinTalpfa.Mesh = sleepers.Commit();

		// Segédvonalak megjelenítése
		var handles = new SurfaceTool();
		handles.Begin(Mesh.PrimitiveType.Lines);

		if (kattAllapot >= 2)
		{
			handles.AddVertex(gorbe[0]);
			handles.AddVertex(gorbe[1]);
		}
		else if (kattAllapot == 1)
		{
			handles.AddVertex(gorbe[0]);
			handles.AddVertex(elonezetPozicio);
		}

		if (kattAllapot >= 4)
		{
			handles.AddVertex(gorbe[2]);
			handles.AddVertex(gorbe[3]);
		}
		else if (kattAllapot == 3)
		{
			handles.AddVertex(gorbe[2]);
			handles.AddVertex(elonezetPozicio);
		}

		sinVonalak.Mesh = handles.Commit();
	}

	private void UpdateInstructions()
	{
		kiir.Text = kattAllapot switch

			1 => "2/4 - Mozgasd az egeret a kezdoiranyhoz!",
			2 => "3/4 - Mozgasd az egeret a vegponthoz!",
			3 => "4/4 - Mozgasd az egeret a vegiranyhoz!",
			_ => "Sín kész! Jobb kattintas: ujrakezdes."
		};
	}

	private void ResetTrack()
	{
		kattAllapot = 0;

	foreach (var marker in segedPont)
		{
			marker.Visible = false;
		}

		sinVas.Mesh = null;
		sinTalpfa.Mesh = null;
		sinVonalak.Mesh = null;

		UpdateInstructions();
	}
